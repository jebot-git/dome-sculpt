  set(paint_compute_source src/paint-composite-cpu.cpp)
elseif(DOME_COMPUTE_BACKEND STREQUAL "CUDA")
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(compute_source src/backend_gpu.cu)
  set(paint_compute_source src/paint-composite-gpu.cu)
elseif(DOME_COMPUTE_BACKEND STREQUAL "ROCM")
  # The HIP Clang link driver may default to PIE while the host C/C++ compiler
  # defaults to absolute code. Keep all project objects linkable across that
  # mixed-driver boundary, while respecting an explicit caller setting.
  if(NOT DEFINED CMAKE_POSITION_INDEPENDENT_CODE)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  endif()
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 20)
  set_source_files_properties(src/backend_gpu.cu src/paint-composite-gpu.cu PROPERTIES LANGUAGE HIP)
  set(compute_source src/backend_gpu.cu)
  set(paint_compute_source src/paint-composite-gpu.cu)
elseif(DOME_COMPUTE_BACKEND STREQUAL "ONEAPI")
  include(CheckCXXSourceCompiles)
  set(dome_saved_required_flags "${CMAKE_REQUIRED_FLAGS}")
  set(CMAKE_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS} -fsycl")
  check_cxx_source_compiles("#include <sycl/sycl.hpp>\nint main() { sycl::queue queue{sycl::gpu_selector_v}; queue.single_task([] {}); queue.wait_and_throw(); }" DOME_HAS_SYCL_TOOLCHAIN)
  set(CMAKE_REQUIRED_FLAGS "${dome_saved_required_flags}")
  unset(dome_saved_required_flags)
  if(NOT DOME_HAS_SYCL_TOOLCHAIN)
    message(FATAL_ERROR "ONEAPI requires a complete SYCL compiler, headers and linkable runtime (for example icpx -fsycl). A compiler that only accepts -fsycl is insufficient; inspect CMakeConfigureLog.yaml.")
  endif()
  set(compute_source src/backend_sycl.cpp)
  set(paint_compute_source src/paint-composite-sycl.cpp)
else()
  message(FATAL_ERROR "Unknown DOME_COMPUTE_BACKEND=${DOME_COMPUTE_BACKEND}")
endif()

# multiply/add boundaries identical to the integer GPU implementation even

  APPEND PROPERTY COMPILE_OPTIONS "$<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off>")

  endif()

  if(NOT dome_fltk_cleanup_sha256 STREQUAL "c40abb454f7cd648227cdaca8802a9f01073b992250accd10969c5b9627537c3")

  foreach(dome_bitmap_package freetype2 harfbuzz)
    pkg_get_variable(dome_bitmap_package_prefix ${dome_bitmap_package} prefix)
    if(NOT dome_bitmap_package_prefix STREQUAL dome_bitmap_prefix)

  foreach(dome_bitmap_target DOME_BITMAP_RUNTIME NATIVEGL)
    foreach(dome_bitmap_library IN LISTS ${dome_bitmap_target}_LINK_LIBRARIES)
      get_filename_component(dome_bitmap_library_name "${dome_bitmap_library}" NAME)
      if(dome_bitmap_library_name MATCHES "^lib(cairo|freetype|harfbuzz)(\\.so(\\..*)?|\\.a)$")
        get_filename_component(dome_bitmap_library_real "${dome_bitmap_library}" REALPATH)
        cmake_path(IS_PREFIX dome_bitmap_prefix "${dome_bitmap_library_real}" NORMALIZE dome_bitmap_library_matches)
        if(NOT dome_bitmap_library_matches)
          message(FATAL_ERROR "${dome_bitmap_target} still resolves ${dome_bitmap_library} outside the selected canonical bitmap runtime ${dome_bitmap_prefix}. Clear *DOME_BITMAP_RUNTIME*, *NATIVEGL* and *RENDERFONTS* cache entries when switching runtimes; see docs/TOOLKIT.md.")
        endif()
      endif()
