option(DOME_BUILD_UI "Build the native FLTK application" ON)

if(DOME_COMPUTE_BACKEND STREQUAL "CPU")
  set(compute_source src/backend_cpu.cpp)
elseif(DOME_COMPUTE_BACKEND STREQUAL "CUDA")
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(compute_source src/backend_gpu.cu)
elseif(DOME_COMPUTE_BACKEND STREQUAL "ROCM")
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 20)
  set_source_files_properties(src/backend_gpu.cu PROPERTIES LANGUAGE HIP)
  set(compute_source src/backend_gpu.cu)
elseif(DOME_COMPUTE_BACKEND STREQUAL "ONEAPI")
  include(CheckCXXSourceCompiles)
  set(dome_saved_required_flags "${CMAKE_REQUIRED_FLAGS}")
  set(CMAKE_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS} -fsycl")
  check_cxx_source_compiles("#include <sycl/sycl.hpp>\nint main() { sycl::queue queue{sycl::gpu_selector_v}; queue.single_task([] {}); queue.wait_and_throw(); }" DOME_HAS_SYCL_TOOLCHAIN)
  set(CMAKE_REQUIRED_FLAGS "${dome_saved_required_flags}")
  unset(dome_saved_required_flags)
  if(NOT DOME_HAS_SYCL_TOOLCHAIN)
    message(FATAL_ERROR "ONEAPI requires a complete SYCL compiler, headers and linkable runtime (for example icpx -fsycl). A compiler that only accepts -fsycl is insufficient; inspect CMakeConfigureLog.yaml.")
  endif()
  set(compute_source src/backend_sycl.cpp)
else()
  message(FATAL_ERROR "Unknown DOME_COMPUTE_BACKEND=${DOME_COMPUTE_BACKEND}")
endif()
