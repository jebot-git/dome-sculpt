cmake_minimum_required(VERSION 3.24)
project(DomeSculpt VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DOME_COMPUTE_BACKEND CPU CACHE STRING "Sculpt compute backend: CPU, CUDA, ROCM, ONEAPI")
set_property(CACHE DOME_COMPUTE_BACKEND PROPERTY STRINGS CPU CUDA ROCM ONEAPI)
option(DOME_BUILD_UI "Build the native FLTK application" ON)
option(DOME_BUILD_XR "Build native OpenXR sculpting" ON)
if(DOME_COMPUTE_BACKEND STREQUAL "CPU")
  set(compute_source src/backend_cpu.cpp)
elseif(DOME_COMPUTE_BACKEND STREQUAL "CUDA")
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(compute_source src/backend_gpu.cu)
elseif(DOME_COMPUTE_BACKEND STREQUAL "ROCM")
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 20)
  set_source_files_properties(src/backend_gpu.cu PROPERTIES LANGUAGE HIP)
  set(compute_source src/backend_gpu.cu)
elseif(DOME_COMPUTE_BACKEND STREQUAL "ONEAPI")
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag(-fsycl DOME_HAS_SYCL)
  if(NOT DOME_HAS_SYCL)
    message(FATAL_ERROR "ONEAPI requires a SYCL compiler. Configure with -DCMAKE_CXX_COMPILER=icpx")
  endif()
  set(compute_source src/backend_sycl.cpp)
else()
  message(FATAL_ERROR "Unknown DOME_COMPUTE_BACKEND=${DOME_COMPUTE_BACKEND}")
endif()
find_package(Threads REQUIRED)
add_library(xatlas STATIC third_party/xatlas/xatlas.cpp)
target_include_directories(xatlas PUBLIC third_party/xatlas)
target_link_libraries(xatlas PRIVATE Threads::Threads)
add_library(dome-sculpt-core src/volume.cpp src/mesh.cpp src/atlas.cpp src/export.cpp src/input.cpp src/document.cpp third_party/openbrush/navigation.cpp third_party/openbrush/controller-profiles.cpp ${compute_source})
target_include_directories(dome-sculpt-core PUBLIC include third_party/openbrush)
target_link_libraries(dome-sculpt-core PRIVATE xatlas Threads::Threads)
target_compile_options(dome-sculpt-core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
if(DOME_COMPUTE_BACKEND STREQUAL "ONEAPI")
  target_compile_options(dome-sculpt-core PRIVATE -fsycl)
  target_link_options(dome-sculpt-core PUBLIC -fsycl)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(dome-sculpt-document-test tests/document.cpp)
  target_link_libraries(dome-sculpt-document-test PRIVATE dome-sculpt-core Threads::Threads)
  add_test(NAME shared-document COMMAND dome-sculpt-document-test)
  add_executable(dome-sculpt-core-test tests/core.cpp)
  target_link_libraries(dome-sculpt-core-test PRIVATE dome-sculpt-core)
  add_test(NAME sculpt-geometry COMMAND dome-sculpt-core-test)
  add_executable(dome-sculpt-gltf-test tests/gltf.cpp)
  target_include_directories(dome-sculpt-gltf-test PRIVATE third_party/cgltf)
  target_link_libraries(dome-sculpt-gltf-test PRIVATE dome-sculpt-core)
  add_test(NAME gltf-interoperability COMMAND dome-sculpt-gltf-test)
endif()
if(DOME_BUILD_UI OR DOME_BUILD_XR)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(NATIVEGL REQUIRED IMPORTED_TARGET egl gl freetype2)
  add_library(dome-sculpt-render src/render.cpp src/palette.cpp third_party/openbrush/graphics.cpp)
  target_link_libraries(dome-sculpt-render PUBLIC dome-sculpt-core PkgConfig::NATIVEGL)
  if(BUILD_TESTING)
    pkg_check_modules(RENDERFONTS REQUIRED IMPORTED_TARGET fontconfig zlib)
    add_executable(dome-sculpt-render-test tests/render.cpp)
    target_include_directories(dome-sculpt-render-test PRIVATE third_party/desktop)
    target_link_libraries(dome-sculpt-render-test PRIVATE dome-sculpt-render PkgConfig::RENDERFONTS)
    add_test(NAME spatial-gpu-render COMMAND dome-sculpt-render-test ${CMAKE_CURRENT_BINARY_DIR}/gpu)
    set_tests_properties(spatial-gpu-render PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1")
  endif()
  if(DOME_BUILD_XR)
    pkg_check_modules(OPENXR REQUIRED IMPORTED_TARGET openxr)
    add_library(dome-sculpt-xr src/xr.cpp)
    target_link_libraries(dome-sculpt-xr PUBLIC dome-sculpt-render PkgConfig::OPENXR)
    if(BUILD_TESTING)
      add_library(dome-sculpt-test-runtime SHARED tests/xr_runtime.cpp)
      target_link_libraries(dome-sculpt-test-runtime PRIVATE PkgConfig::NATIVEGL)
      if(DOME_BUILD_UI)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME openxr-client COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/xr_runtime.py $<TARGET_FILE:dome-sculpt> $<TARGET_FILE:dome-sculpt-test-runtime> ${CMAKE_CURRENT_BINARY_DIR}/xr-client)
      endif()
    endif()
  endif()
endif()
if(DOME_BUILD_UI)
  list(PREPEND CMAKE_PREFIX_PATH /opt/fltk-vr/toolkit/share/fltk)
  find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
  find_package(OpenGL REQUIRED)
  find_library(DOME_GLU NAMES GLU libGLU.so.1 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(APPEARANCE REQUIRED IMPORTED_TARGET fontconfig zlib)
  add_executable(dome-sculpt src/app.cpp)
  target_include_directories(dome-sculpt PRIVATE third_party/desktop)
  target_link_libraries(dome-sculpt PRIVATE dome-sculpt-core dome-sculpt-render fltk::fltk-shared fltk::images-shared fltk::gl-shared OpenGL::GL ${DOME_GLU} PkgConfig::APPEARANCE)
  if(DOME_BUILD_XR)
    target_link_libraries(dome-sculpt PRIVATE dome-sculpt-xr)
    target_compile_definitions(dome-sculpt PRIVATE DOME_HAS_XR=1)
  endif()
  include(third_party/desktop/desktop-menu.cmake)
  de_xr_desktop_menu(dome-sculpt)
  set_target_properties(dome-sculpt PROPERTIES BUILD_RPATH /opt/fltk-vr/toolkit/lib INSTALL_RPATH /opt/fltk-vr/toolkit/lib)
  install(TARGETS dome-sculpt RUNTIME DESTINATION bin)
endif()
