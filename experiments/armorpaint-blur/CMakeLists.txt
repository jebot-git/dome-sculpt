cmake_minimum_required(VERSION 3.24)
project(ArmorPaintBlurProof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DOME_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(CMAKE_CURRENT_BINARY_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  message(FATAL_ERROR "Use an out-of-source build directory")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${generated}/blur_shader.hpp" "${generated}/blur.frag"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_shader.py" --output-dir "${generated}"
  DEPENDS generate_shader.py vendor/make_blur.c
  VERBATIM)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GL REQUIRED IMPORTED_TARGET opengl egl)
add_library(armorpaint-blur STATIC blur_adapter.cpp "${generated}/blur_shader.hpp")
target_include_directories(armorpaint-blur PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(armorpaint-blur PRIVATE "${generated}")
target_link_libraries(armorpaint-blur PUBLIC PkgConfig::GL)
target_compile_options(armorpaint-blur PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(blur-proof proof.cpp "${DOME_SOURCE}/third_party/openbrush/graphics.cpp")
target_include_directories(blur-proof PRIVATE "${DOME_SOURCE}/third_party/openbrush")
target_link_libraries(blur-proof PRIVATE armorpaint-blur)
target_compile_options(blur-proof PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(api glCreateShader glDeleteShader glCreateProgram glDeleteProgram glGenFramebuffers glDeleteFramebuffers glGenVertexArrays glDeleteVertexArrays)
  target_link_options(blur-proof PRIVATE "-Wl,--wrap=${api}")
endforeach()
option(PROOF_SANITIZE "Sanitize isolated native adapter/consumer" OFF)
if(PROOF_SANITIZE)
  foreach(target armorpaint-blur blur-proof)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
